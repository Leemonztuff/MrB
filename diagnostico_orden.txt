select 'COLUMN' as tipo, table_name || '.' || ordinal_position::text || '.' || column_name as nombre,
       data_type || coalesce(' / ' || udt_name, '') || ' / null=' || is_nullable || ' / def=' || coalesce(column_default, '-') as detalle
from information_schema.columns
where table_schema = 'public'

union all

select 'ENUM', t.typname, string_agg(e.enumlabel, ', ' order by e.enumsortorder)
from pg_type t
join pg_enum e on e.enumtypid = t.oid
join pg_namespace n on n.oid = t.typnamespace
where n.nspname = 'public'
group by t.typname

union all

select 'FUNC', p.proname || '(' || pg_get_function_identity_arguments(p.oid) || ')', pg_get_functiondef(p.oid)
from pg_proc p
join pg_namespace n on n.oid = p.pronamespace
where n.nspname = 'public'

union all

select 'INDEX', indexname, indexdef
from pg_indexes
where schemaname = 'public'

union all

select 'POLICY', policyname, tablename || ' / roles=' || roles::text || ' / using=' || coalesce(qual, '-') || ' / check=' || coalesce(with_check, '-')
from pg_policies
where schemaname = 'public'

order by 1, 2;
