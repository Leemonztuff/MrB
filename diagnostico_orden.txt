select 'FUNCION' as seccion, pg_get_functiondef(oid) as valor
from pg_proc
where proname = 'create_order_with_items'

union all

select 'ENUM payment_type', string_agg(e.enumlabel, ', ' order by e.enumsortorder)
from pg_type t
join pg_enum e on e.enumtypid = t.oid
where t.typname = (
    select udt_name
    from information_schema.columns
    where table_schema = 'public'
      and table_name = 'orders'
      and column_name = 'payment_type'
);
